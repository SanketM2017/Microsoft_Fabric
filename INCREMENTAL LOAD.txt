

	SESSION: HOW TO COPY MULTIPLE TABLES FROM SQL DB TO ADLS GEN2 USING ADF  --> INCREMENTAL LOAD
	
	TYPES OF PIPELINE :- 1) FULL LOAD PIPELINE	 --> 
						 2) INCREMENTAL LOAD PIPELINE
						 
						 
						 
						 
	1)	FIRST LOOK-UP ACTIVITY -- TO READ ALL CONFIG TABLE DETAILS
	
		 CONFIG TABLE AS BELOW 
		 
		 create table tblconfig(
				id int,
				srcschema varchar(50),
				srctable varchar(50),
				sinkcontainer varchar(50),
				sinkfolder varchar(250),
				sink_file_name varchar(20),
				lpv varchar(50),
				deltacol varchar(50),
				is_active int);

				insert into tblconfig values 
				(1,'dbo','cust','global','incremental_data','cust.csv','0','cid',1),
				(2,'dbo','trans','global','incremental_data','trans.csv','2024-01-01 10:10:10','transdate',1),
				(3,'dbo','orders','global','incremental_data','orders.csv','1900-01-01 10:10:10','orderdate',1),
				(4,'dbo','emp','global','incremental_data','emp.csv','10','eid',0)
	
	2)	PASS THIS METADATA TO FOREACH ACTIVITY USING EXPRESSION --> @activity('Lookup1').output.value
	
	3)	INSIDE THE FOREACH USE 2 LOOKUP ACTIVITY TO FIND MAX VALUES FOR EACH TABLES AND ENABLE FIRSTROW ONLY AND USE QUERY OPTION 
	
	4)	USING BELOW QUERY --> SELECT MAX(DELTACOL) AS MAX_VALUE FROM TABLE_NAME
		DYNAMIC QUERY -->
	5)	select max(@{item().deltacol}) as max_value from @{item().srctable}
	
	6) AFTER LOOKUP2 USE IF CONDITION  --> IF CONDITION ACTIVITY IS USED TO CHECK LPV AND MAX_VALUES ARE MATCHING OR NOT
	
				A dynamic boolean expression (e.g. @equals(...), @contains(...), or value comparisons).
				Two activity groups:

				--> if True Activities (executed when expression is true).

				--> if False Activities (executed when expression is false)

	7) IN IF CONDITION IN SETTING IN EXPRESSION USE BELOW EXPRESSION
		@equals(item().lpv,activity('Find Max Value').output.firstRow.max_value)
		( IT WILL CHECK LPV AND MAX VALUE ARE MATCHING OR NOT 
				IF ITS MATCHING MEANS NO NEW RECORD ARE UPDATED IN THE TABLE
				IF NOT MATCHING MEANS TABLE IS UPDATED -- SO CONDITIPN BECOMES FALSE AND IN FALSE SIDE USE COPY DATA ACTIVITY
		
	8) COPY DATA ACTIVITY --> USE SAME DATASET AND LS THAT WE JAVE USED FOR BOTH LOOKUP ACTIVITY
	
	IN SOURCE SIDE SELECT THE QUERY OPTION AND MENTION BELOW EXPRESSION
	
		--> SELECT * FROM SCHEMA.TABLE_NAME WHERE DELTACOL>LPV;
		SAMPLE QUERY 	--> SELECT * FROM DBO.CUST WHERE CID>0;
		EXPRESSION 		==> select * from @{item().srcschema}.@{item().srctable} where @{item().deltacol} > '@{item().lpv}'
		
	IN SINK SIDE USE DATASET AND LS FOR ADLS GEN2 AND CREATE 3 DATASET LEVEL PARAMETERS FOR
			SINK_CONTAINER_NAME
			SINK_FOLDER_NAME
			SINK_FILE_NAME  --> in dataset properties while passing this dynamic use concat function to generate the yyyy/mm/dd folder
			
			@concat(formatDateTime(utcNow(),'yyyy/MM/dd'),'/',dataset().sink_file_name)
			
				-- so it will create a folder like 2025/08/05/file_name
			
	GOTO PIPELINE AND IN COPY ACTIVITY AT SINK SIDE USE BELOW EXPRESSIONS
	
			@item().SINK_CONTAINER_NAME
			@item().SINK_FOLDER_NAME
			@item().SINK_FILE_NAME
	
	9) AFTER COPY DATA ACTIVITY USE STORED PROCEDURE ACTIVITY TO UPDATE THE CONFIG TABLE IN SQL DB SOURCE
	
	
					create proc tblconfig_usp
					@lpv_param varchar(50),
					@id int
					as
					BEGIN
						update tblconfig set lpv=@lpv_param where id=@id
					END
